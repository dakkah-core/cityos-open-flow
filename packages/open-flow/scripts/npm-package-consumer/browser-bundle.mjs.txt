import assert from 'node:assert/strict'
import { writeFile } from 'node:fs/promises'
import path from 'node:path'
import { fileURLToPath } from 'node:url'
import { build, version } from 'vite'

assert.equal(version, '7.3.6', 'This fixture intentionally exercises the Vite 7 Rollup consumer.')
const root = fileURLToPath(new URL('.', import.meta.url))
const directory = process.argv[2]
let esbuildOptions
let format

// Keep default Rollup chunking/treeshaking and Vite CSS processing. Only the
// final JavaScript esbuild pass moves to a fresh process after this one closes.
const result = await build({
  configFile: false,
  root,
  logLevel: 'error',
  plugins: [
    {
      name: 'open-flow-browser-inventory',
      configResolved(config) {
        assert.equal(config.build.target, 'esnext')
        assert.equal(config.build.lib, false)
        assert.equal(config.build.cssMinify, true)
        assert.deepEqual(Object.keys(config.esbuild).sort(), ['charset', 'jsxDev', 'legalComments'])
        esbuildOptions = config.esbuild
      },
      writeBundle(options) {
        assert.equal(options.format, 'es')
        format = options.format
      },
    },
  ],
  build: {
    target: 'esnext',
    minify: false,
    // Vite otherwise derives false from minify:false; the original used true.
    cssMinify: true,
    outDir: directory,
    emptyOutDir: false,
    rollupOptions: { input: fileURLToPath(new URL('./preview.mjs', import.meta.url)) },
  },
})
assert.equal(format, 'es')
const chunks = (Array.isArray(result) ? result : [result])
  .flatMap((bundle) => bundle.output)
  .filter((output) => output.type === 'chunk')
  .map(({ fileName, isDynamicEntry }) => ({ fileName, isDynamicEntry }))
assert.ok(chunks.some((chunk) => chunk.isDynamicEntry), 'Missing lazy browser chunks.')

// Effective options from installed Vite 7.3.6 config.js:
// resolveEsbuildTranspileOptions (non-library es/esnext, minify:'esbuild') plus
// transformWithEsbuild (including its internal sourcemap/JS tsconfig defaults).
await writeFile(
  path.join(directory, 'inventory.json'),
  JSON.stringify({
    viteVersion: version,
    options: {
      sourcemap: true,
      ...esbuildOptions,
      loader: 'js',
      target: 'esnext',
      format: 'esm',
      supported: { 'dynamic-import': true, 'import-meta': true },
      minify: true,
      treeShaking: true,
      tsconfigRaw: { compilerOptions: { useDefineForClassFields: false } },
    },
    chunks,
  }),
)
console.log(`[npm-package:browser] bundle:inventory chunks=${chunks.length}`)
