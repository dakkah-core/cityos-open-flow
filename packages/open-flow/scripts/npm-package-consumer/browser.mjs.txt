import { spawn } from 'node:child_process'
import { mkdtemp } from 'node:fs/promises'
import path from 'node:path'
import { fileURLToPath } from 'node:url'
import { joinBrowserProcess } from './browser-process.mjs'

const root = fileURLToPath(new URL('.', import.meta.url))
const directory = await mkdtemp(path.join(root, 'browser-output-'))
// Use native Bun's lower-memory GC mode for bounded build children. Node
// fixtures retain Node's own runtime flags.
const runtimeArgs = process.versions.bun ? ['--smol'] : []

// The outer consumer owns cleanup through assertTemporaryConsumerTarget.
// Neither this parent nor its process-join helper imports Vite or esbuild.
for (const phase of ['bundle', 'minify']) {
  console.log(`[npm-package:browser] ${phase}:start`)
  const child = spawn(process.execPath, [...runtimeArgs, fileURLToPath(new URL(`./browser-${phase}.mjs`, import.meta.url)), directory], {
    cwd: root,
    stdio: 'inherit',
    windowsHide: true,
  })
  await joinBrowserProcess(child, phase)
  console.log(`[npm-package:browser] ${phase}:pass`)
}
